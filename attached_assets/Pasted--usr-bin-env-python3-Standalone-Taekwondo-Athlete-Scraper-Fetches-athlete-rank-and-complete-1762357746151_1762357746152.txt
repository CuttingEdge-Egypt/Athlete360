#!/usr/bin/env python3
"""
Standalone Taekwondo Athlete Scraper
Fetches athlete rank and complete competitive history by name and country
No dependencies on other project files - completely self-contained
"""

import json
import logging
from datetime import datetime
from typing import Dict, List, Optional, Any
import cloudscraper

# Set up logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


def get_athlete_data(name: str, country: str, month: int = None, year: int = None) -> Dict[str, Any]:
    """
    Standalone function to scrape athlete's rank and complete competitive history
    by athlete name and country.
    
    Args:
        name: Athlete's name (partial match supported)
        country: Athlete's country
        month: Month for rankings (1-12, where 1=January). Defaults to current month
        year: Year for rankings. Defaults to current year
        
    Returns:
        Dictionary containing:
        - success: bool
        - athlete_name: str
        - country: str
        - userid: str
        - wtf_license_id: str
        - gender: str
        - current_rank: int
        - points: float
        - weight_division: str
        - profile: dict (complete profile info)
        - competitions: list (all competitions with details)
        - total_competitions: int
        - ranking_categories: list (all ranking categories)
        
    Example:
        result = get_athlete_data("Jun JANG", "Republic of Korea")
        if result['success']:
            print(f"Athlete: {result['athlete_name']}")
            print(f"Rank: #{result['current_rank']}")
            print(f"Total Competitions: {result['total_competitions']}")
    """
    
    # Use current date if not specified
    if month is None:
        month = datetime.now().month
    if year is None:
        year = datetime.now().year
    
    # Convert month to 0-indexed (API expects 0=Jan, 11=Dec)
    api_month = month - 1
    
    logger.info(f"Searching for athlete: {name} from {country}")
    logger.info(f"Time period: {datetime(year, month, 1).strftime('%B %Y')}")
    
    try:
        # Step 1: Search for athlete in rankings to get userid
        logger.info("Step 1: Searching rankings database...")
        athletes = _search_athlete_in_rankings(name, country, api_month, year)
        
        if not athletes:
            logger.warning(f"No athlete found matching: {name} from {country}")
            return {
                'success': False,
                'error': 'Athlete not found in rankings',
                'searched_name': name,
                'searched_country': country
            }
        
        # Get the first matching athlete
        athlete = athletes[0]
        userid = athlete.get('userid')
        
        if not userid:
            logger.warning("Athlete found but no userid available")
            return {
                'success': False,
                'error': 'Athlete found but userid not available',
                'athlete_data': athlete
            }
        
        logger.info(f"Found athlete: {athlete.get('name')} (userid: {userid})")
        
        # Step 2: Get comprehensive athlete data including competitions
        logger.info("Step 2: Fetching comprehensive athlete data...")
        result = _get_complete_athlete_data(userid, year, api_month)
        
        if not result.get('success'):
            return {
                'success': False,
                'error': result.get('error', 'Failed to fetch athlete data'),
                'ranking_data': athlete
            }
        
        # Combine ranking and competition data
        profile = result.get('profile', {})
        competitions = result.get('competitions', [])
        ranking_categories = result.get('ranking_categories', [])
        
        return {
            'success': True,
            'athlete_name': f"{profile.get('preferred_first_name', '')} {profile.get('preferred_last_name', '')}".strip() or athlete.get('name'),
            'country': profile.get('country') or athlete.get('country'),
            'userid': userid,
            'wtf_license_id': profile.get('wtf_license_id'),
            'gender': profile.get('gender'),
            'current_rank': athlete.get('ranking'),
            'points': athlete.get('points'),
            'weight_division': athlete.get('weight_division'),
            'change': athlete.get('change'),
            'profile': profile,
            'competitions': competitions,
            'total_competitions': len(competitions),
            'ranking_categories': ranking_categories,
            'search_period': {
                'month': month,
                'year': year
            }
        }
        
    except Exception as e:
        logger.error(f"Error during athlete search: {str(e)}")
        return {
            'success': False,
            'error': str(e),
            'searched_name': name,
            'searched_country': country
        }


def _search_athlete_in_rankings(name: str, country: str, month: int, year: int) -> List[Dict]:
    """Search for athlete in rankings API"""
    
    # Country UUID mappings (abbreviated - add more as needed)
    COUNTRY_MAPPINGS = {
        "Republic of Korea": "11e6eef3-fe5f-9851-92b4-12f817a4f090",
        "Egypt": "11e6eef3-fe4e-a833-92b4-12f817a4f090",
        "Italy": "11e6eef3-fe58-936a-92b4-12f817a4f090",
        "United States of America": "11e6eef3-fe3b-1fe0-92b4-12f817a4f090",
        "Great Britain": "11e6eef3-fe41-3a79-92b4-12f817a4f090",
        "France": "11e6eef3-fe3d-42c8-92b4-12f817a4f090",
        "Germany": "11e6eef3-fe3d-b7f9-92b4-12f817a4f090",
        "Spain": "11e6eef3-fe40-7725-92b4-12f817a4f090",
        "Brazil": "11e6eef3-fe3c-3154-92b4-12f817a4f090",
        "Mexico": "11e6eef3-fe3e-f07e-92b4-12f817a4f090",
        "Canada": "11e6eef3-fe3c-7f75-92b4-12f817a4f090",
        "Australia": "11e6eef3-fe3b-6e01-92b4-12f817a4f090",
        "Japan": "11e6eef3-fe3e-7b4c-92b4-12f817a4f090",
        # Add more countries as needed
    }
    
    # Create cloudscraper session
    session = cloudscraper.create_scraper(
        browser={'browser': 'chrome', 'platform': 'windows', 'desktop': True}
    )
    
    session.headers.update({
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36',
        'Accept': 'application/json, text/plain, */*',
    })
    
    # Build API URL
    api_url = "https://worldtkd.simplycompete.com/rankingsV2"
    params = [
        "limit=100",
        f"month={month}",
        f"year={year}",
        "rankingTypeId=11ef3b4e-05ce-797c-aca4-064aef8133e9",  # World Kyorugi Rankings
        "subCategory1=11ef3b4e-3058-36ed-aca4-064aef8133e9",   # World Senior Division
    ]
    
    # Add country filter if available
    if country in COUNTRY_MAPPINGS:
        params.append(f"countryId={COUNTRY_MAPPINGS[country]}")
    
    url = f"{api_url}?{'&'.join(params)}"
    
    # Fetch data
    response = session.get(url, timeout=30)
    response.raise_for_status()
    
    data = response.json()
    athletes = []
    
    # Extract athletes from response
    if isinstance(data, list):
        for item in data:
            athlete = _extract_athlete_from_json(item)
            if athlete and name.lower() in athlete.get('name', '').lower():
                athletes.append(athlete)
    
    return athletes


def _extract_athlete_from_json(item: Dict) -> Optional[Dict]:
    """Extract athlete data from JSON item"""
    try:
        ranking = item.get('rank', 0)
        name = item.get('name', '')
        country = item.get('nation', '')
        points = item.get('points', 0)
        userid = item.get('userId', '')
        wtf_license_id = item.get('wtfLicenseId', '')
        
        # Determine gender
        gender_id = item.get('gender', '')
        gender = ""
        if gender_id == "11e6eef3-fd24-a707-92b4-12f817a4f090":
            gender = "Male"
        elif gender_id == "11e6eef3-fd24-ce18-92b4-12f817a4f090":
            gender = "Female"
        
        # Calculate rank change
        current_rank = int(ranking) if ranking else 0
        previous_rank = item.get('previousRank', 0)
        change = ""
        if previous_rank and current_rank:
            prev_rank = int(previous_rank)
            if prev_rank > current_rank:
                change = f"+{prev_rank - current_rank}"
            elif prev_rank < current_rank:
                change = f"-{current_rank - prev_rank}"
            else:
                change = "0"
        
        return {
            'ranking': str(ranking),
            'name': str(name).strip(),
            'country': str(country).strip(),
            'gender': gender,
            'weight_division': 'N/A',
            'points': str(points),
            'change': change,
            'userid': str(userid) if userid else '',
            'wtfLicenseId': str(wtf_license_id) if wtf_license_id else '',
        }
    except Exception as e:
        logger.warning(f"Error processing JSON item: {e}")
        return None


def _get_complete_athlete_data(user_id: str, year: int, month: int) -> Dict[str, Any]:
    """Get complete athlete profile and competition data"""
    
    # Create cloudscraper session
    session = cloudscraper.create_scraper(
        browser={'browser': 'chrome', 'platform': 'windows', 'desktop': True}
    )
    
    session.headers.update({
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36',
        'Accept': 'application/json, text/plain, */*',
        'Referer': 'https://worldtkd.simplycompete.com/',
    })
    
    # API endpoint
    api_url = "https://worldtkd.simplycompete.com/getPlayerProfileV2"
    
    params = {
        'countryId': 'null',
        'month': str(month),
        'rankingTypeId': '11ef3b4e-05ce-797c-aca4-064aef8133e9',
        'searchQuery': '',
        'subCategory1': '11ef3b4e-3058-36ed-aca4-064aef8133e9',
        'userId': user_id,
        'year': str(year)
    }
    
    try:
        response = session.get(api_url, params=params, timeout=30)
        response.raise_for_status()
        
        api_data = response.json()
        data_section = api_data.get('data', {})
        data_data = data_section.get('data', {})
        
        # Extract profile
        profile = {
            'preferred_first_name': data_data.get('preferredFirstName', ''),
            'preferred_last_name': data_data.get('preferredLastName', ''),
            'first_name': data_data.get('firstName', ''),
            'last_name': data_data.get('lastName', ''),
            'wtf_license_id': data_data.get('wtfLicenseId', ''),
            'profile_pic_url': data_data.get('profilePicUrl', ''),
            'country': data_data.get('country', ''),
            'gender': data_data.get('gender', ''),
            'birth_year': data_data.get('birthYear', ''),
            'age': data_data.get('age', ''),
        }
        
        # Extract competitions
        competitions = _extract_competitions(data_data)
        ranking_categories = data_data.get('rankingCategory', [])
        
        return {
            'success': True,
            'profile': profile,
            'competitions': competitions,
            'ranking_categories': ranking_categories,
        }
        
    except Exception as e:
        logger.error(f"Error fetching athlete data: {str(e)}")
        return {
            'success': False,
            'error': str(e)
        }


def _extract_competitions(data_data: Dict) -> List[Dict]:
    """Extract all competitions from athlete data"""
    competitions = []
    
    ranking_categories = data_data.get('rankingCategory', [])
    
    for category in ranking_categories:
        category_name = category.get('rankingCategoryName', 'Unknown')
        event_results = category.get('eventResults', [])
        
        for event in event_results:
            competition = {
                'event': event.get('eventName', 'Unknown Event'),
                'g_rank': event.get('gRank', 'N/A'),
                'date': event.get('endDate', 'N/A'),
                'location': event.get('location', 'N/A'),
                'place': event.get('place', 'N/A'),
                'points': str(event.get('rankingPoints', 'N/A')),
                'category': category_name,
                'event_id': event.get('eventId', ''),
                'event_result': event.get('eventResult', 'N/A')
            }
            competitions.append(competition)
    
    # Sort by date (most recent first)
    competitions.sort(key=lambda x: x.get('date', ''), reverse=True)
    
    return competitions
